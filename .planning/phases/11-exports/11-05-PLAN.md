    interface SearchParams { from?: string; to?: string; project?: string; }

    export default async function ExportsPage({ searchParams }: { searchParams: Promise<SearchParams> }) {
      const session = await auth();        // D-114 page-level: redirect on null (matches hakedis pattern)
      if (!session) redirect('/auth/signin');

      const sp = await searchParams;
      const t = await getTranslations('dashboard.admin.exports');
      const locale = await getLocale();

      // Parallel data fetch — both queries are tenant-scoped via getDefaultTenantId()
      const [projects, periods] = await Promise.all([
        getProjects(),
        getAllFinishedPeriods(),
      ]);
      // FilterBar requires projectOptions + personOptions in the same shape as hakedis page.
      // For the Exports hub, we don't surface a person filter (per UI-SPEC §"Global FilterBar" —
      // date-range + project only). Pass [] for personOptions; FilterBar handles the empty array.
      const projectOptions = projects.map(p => ({ value: p.id, label: p.name }));

      // Build the active-filter chip text
      const activeProject = sp.project ? projects.find(p => p.id === sp.project)?.name : undefined;
      const fromStr = sp.from ?? '';
      const toStr = sp.to ?? '';
      const chip = `${activeProject ?? t('section_ledger')} · ${fromStr || '—'} ${toStr ? '→ ' + toStr : ''}`;

      // Build the URL-encoded query string to forward to download handlers
      const qs = new URLSearchParams();
      if (sp.from) qs.set('from', sp.from);
      if (sp.to) qs.set('to', sp.to);
      if (sp.project) qs.set('project', sp.project);
      const qsString = qs.toString() ? '?' + qs.toString() : '';

      return (
        <div className="space-y-6">
          <div className="space-y-1">
            <h1 className="text-xl font-semibold">{t('heading')}</h1>
            <p className="text-sm text-muted-foreground">{t('subtitle')}</p>
          </div>

          <Suspense fallback={<div className="h-9 w-full animate-pulse rounded-md bg-muted" />}>
            <FilterBar projectOptions={projectOptions} personOptions={[]} />
          </Suspense>

          {/* Submission Ledger trigger card */}
          <Card>
            <CardHeader className="px-4 pt-4 pb-2">
              <div className="flex items-center gap-2">
                <FileSpreadsheet className="h-4 w-4 text-muted-foreground" aria-hidden="true" />
                <CardTitle className="text-sm font-semibold">{t('section_ledger')}</CardTitle>
              </div>
            </CardHeader>
            <CardContent className="px-4 pb-4 flex items-center justify-between gap-4">
              <p className="text-sm text-muted-foreground">{chip}</p>
              <a href={`/api/exports/submissions${qsString}`} download>
                <Button variant="default" size="sm">
                  <Download className="h-4 w-4 mr-1" aria-hidden="true" />
                  {t('download_excel')}
                </Button>
              </a>
            </CardContent>
          </Card>

          {/* Performance Summary trigger card — identical structure */}
          <Card>
            <CardHeader className="px-4 pt-4 pb-2">
              <div className="flex items-center gap-2">
                <FileSpreadsheet className="h-4 w-4 text-muted-foreground" aria-hidden="true" />
                <CardTitle className="text-sm font-semibold">{t('section_performance')}</CardTitle>
              </div>
            </CardHeader>
            <CardContent className="px-4 pb-4 flex items-center justify-between gap-4">
              <p className="text-sm text-muted-foreground">{chip}</p>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser → /dashboard/exports | unauthenticated request — auth() page-level redirect |
| searchParams → query forwarding | from/to/project flow into the download URL query string |
| getAllFinishedPeriods → table render | tenant-scoped data; status filter enforced server-side |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-05-AUTH | Spoofing | /dashboard/exports page | mitigate | `auth()` FIRST statement in the page RSC; redirect to /auth/signin on null. Defense-in-depth: route handlers themselves are independently auth-guarded (D-114). |
| T-11-05-IDOR | Information Disclosure | period picker tenant leak | mitigate | getAllFinishedPeriods is tenant-scoped (verified in Plan 11-01b Task 2). Page does not inline its own SQL. |
| T-11-05-DRAFT | Tampering | export of draft period via picker | mitigate | getAllFinishedPeriods has `status != 'draft'` WHERE clause (Plan 11-01b) — draft periods never appear in the picker. Route handlers also reject draft per Plan 11-04 (defense in depth). |
| T-11-05-XSS | Tampering | project name in chip + periodNumber in table | mitigate | React JSX auto-escapes all interpolated values. No `dangerouslySetInnerHTML`. |
| T-11-05-OPEN-REDIRECT | Tampering | href values | accept | All hrefs are hardcoded `/api/exports/...` patterns + `period.id` (server-fetched UUID). No user-controlled URL building. |
| T-11-05-FILTER-INJ | Tampering | searchParams forwarded to download URL | mitigate | `URLSearchParams` correctly escapes any malicious payload. Receiving route handler re-validates each parameter (Plan 02 + Plan 03 validate Date + UUID). |
</threat_model>
